  private function extrairDados($codigo)
    {
        $dados = [];

        // 🧹 Normalizar texto
        $codigo = trim($codigo);
        $codigo = mb_convert_encoding($codigo, 'UTF-8', 'UTF-8');
        $codigo = preg_replace('/[[:^print:]]/', '', $codigo); // remove bytes de controlo invisíveis

        // 🔄 Corrigir variantes de cedilha e símbolos
        $codigoCorrigido = str_ireplace(
            ['Ã§', 'Â§', 'A§', 'a§', 'Ç', 'ç'],
            'ç',
            $codigo
        );

        // 🧩 Regex mais tolerante:
        // - aceita qualquer carácter não numérico entre A e os 9 dígitos
        // - ignora maiúsculas/minúsculas
        preg_match('/A[^0-9]?(\d{9})/iu', $codigoCorrigido, $matchA);

        // - G seguido de qualquer símbolo (como “ç” ou espaço)
        // - capta até ao próximo “(”
        preg_match('/G[^0-9A-Za-z]?([A-Za-z0-9\s\-\._]+)/iu', $codigoCorrigido, $matchG);

        $dados['A'] = isset($matchA[1]) ? trim($matchA[1]) : null;
        $dados['G'] = isset($matchG[1]) ? trim(str_replace(' ', '-', $matchG[1])) : null;


        return $dados;
    }
	
	  public function enviarCodigo($conf, $tabela, $tabela1, Request $request)
    {
        // 🔹 Obter o código QR do request
        $codigo = $request->input('code');

        // 🔹 Extrair dados com a função auxiliar
        $dados = $this->extrairDados($codigo);
        $A = $dados['A']; // NIF do comerciante
        $G = $dados['G']; // Número da fatura

        // 🔹 Normalizar o número da fatura (remove caracteres estranhos)
        $G = strtoupper(trim(preg_replace('/[^A-Z0-9\-\/]/i', '', $G)));

        // 🔹 Extrair data e valor do QR
        preg_match('/F[ÇC](\d{8})/', $codigo, $matchesData);
        preg_match('/O[ÇC][\(\*]?([\d.,]+)/', $codigo, $matchesValor);

        $data = $matchesData[1] ?? null;
        $valor = isset($matchesValor[1]) ? str_replace(',', '.', $matchesValor[1]) : null;

        // 🔹 Extrair partes da data
        $ano = $data ? substr($data, 0, 4) : null;
        $mes = $data ? substr($data, 4, 2) : null;
        $dia = $data ? substr($data, 6, 2) : null;

        // 🔹 Cria a conexão com a base de dados
        $customConnection = $this->createCustomDatabaseConnection($conf);

        // 🔹 Procurar fatura na BD (uso de LIKE para tolerar diferenças no formato)
        $result = $customConnection->select("
    SELECT fatura.*, artigos_ref.*
    FROM {$tabela}
    LEFT JOIN artigos_ref ON {$tabela}.artigoL1 = artigos_ref.id_artigo
    WHERE fatura.nif_comerciante = ?
    AND REPLACE(UPPER(fatura.num_fatura), '-', '') = REPLACE(?, '-', '')
", [$A, strtoupper($G)]);
        
        // 🔹 Obter lista de artigos ativos
        $result1 = $customConnection->select("
        SELECT artigos_ref.id_artigo, artigos_ref.descricao
        FROM artigos_ref
        WHERE artigos_ref.Ativo = -1
        ORDER BY artigos_ref.descricao
    ");

        // 🔹 Se encontrou a fatura
        if (count($result) > 0) {
            foreach ($result as $row) {
                $numerador = $row->numerador;

                // Se o numerador estiver vazio ou 0 → gerar novo
                if (empty($numerador)) {
                    // Tenta obter o maior numerador do mesmo ano
                    $ultimoNumerador = $customConnection->table($tabela)
                        ->whereRaw('YEAR(data_emissao) = ?', [$ano])
                        ->max('numerador');

                    // Se não encontrar, procura o maior numerador geral
                    if (empty($ultimoNumerador)) {
                        $ultimoNumerador = $customConnection->table($tabela)->max('numerador');
                    }

                    // Incrementa corretamente
                    $numerador = $ultimoNumerador ? $ultimoNumerador + 1 : 1;

                    // Verifica se a fatura já existe
                    $existeFatura = $customConnection->table($tabela)
                        ->where('num_fatura', $row->num_fatura)
                        ->where('nif_comerciante', $A)
                        ->where('nif_consumidor', str_replace('_', '', $conf))
                        ->exists();

                    if ($existeFatura) {
                        // Atualiza o numerador
                        $customConnection->table($tabela)
                            ->where('num_fatura', $row->num_fatura)
                            ->where('nif_comerciante', $A)
                            ->where('nif_consumidor', str_replace('_', '', $conf))
                            ->update(['numerador' => $numerador]);
                    } else {
                        // Caso não exista, insere novo registo
                        $customConnection->table($tabela)->insert([
                            'nif_comerciante' => $A,
                            'num_fatura' => $row->num_fatura,
                            'nif_consumidor' => str_replace('_', '', $conf),
                            'numerador' => $numerador
                        ]);
                    }

                    // Atualiza o numerador para exibição imediata
                    $row->numerador = $numerador;
                }
            }
        } else {
            // ❗ Caso não encontre nenhuma fatura correspondente
            return redirect()->back()->with('error', 'Nenhum dado encontrado para o código QR fornecido.');
        }

        // 🔹 Retorna a view com os dados processados
        return view('validacao.layout.dashboard', [
            'user' => $request->user(),
            'nifatual' => str_replace('_', '', $conf),
            'conf' => $conf,
            'tabela' => $tabela,
            'tabela1' => $tabela1,
            'codigo' => $codigo,
            'result' => $result,
            'result1' => $result1,
            'ano' => $ano,
            'mes' => $mes,
            'dia' => $dia,
            'valor' => $valor,
            'nif_comerciante' => $A,
            'success' => 'Qrcode guardado com sucesso!'
        ]);
    }
